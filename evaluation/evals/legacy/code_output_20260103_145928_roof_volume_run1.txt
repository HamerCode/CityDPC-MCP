import math

def triangle_volume_from_poslist(poslist_str):
    nums = [float(x) for x in poslist_str.strip().split()]
    # group into points, ignoring if last duplicates first
    pts = []
    for i in range(0,len(nums),3):
        pts.append((nums[i], nums[i+1], nums[i+2]))
    # Remove duplicate last point if equal to first
    if len(pts) > 1 and all(abs(pts[0][i]-pts[-1][i])<1e-6 for i in range(3)):
        pts = pts[:-1]
    # Ensure triangle
    assert len(pts)==3, f"Not triangle: {len(pts)}"
    (x1,y1,z1),(x2,y2,z2),(x3,y3,z3) = pts
    # plan area
    area = abs(x1*(y2-y3) + x2*(y3-y1) + x3*(y1-y2)) / 2.0
    base = min(z1,z2,z3)
    avg_diff = ((z1-base)+(z2-base)+(z3-base))/3.0
    vol = area * avg_diff
    return area, avg_diff, vol

def poly_to_triangles(poslist_str):
    nums = [float(x) for x in poslist_str.strip().split()]
    pts=[]
    for i in range(0,len(nums),3): pts.append((nums[i],nums[i+1],nums[i+2]))
    # remove duplicate end if equals first
    if len(pts)>1 and all(abs(pts[0][i]-pts[-1][i])<1e-6 for i in range(3)):
        pts=pts[:-1]
    # fan triangulation from 0
    if len(pts)<3: return []
    tris=[]
    for i in range(1,len(pts)-1):
        tris.append((pts[0], pts[i], pts[i+1]))
    return tris

def volume_poly(poslist):

    total=0.0
    for tri in tris:
        (x1,y1,z1),(x2,y2,z2),(x3,y3,z3)=tri

        base=min(z1,z2,z3)
        avg=((z1-base)+(z2-base)+(z3-base))/3.0
        total += area*avg
    return total
